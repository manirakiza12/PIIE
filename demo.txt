// ===== TRANSCRIPT SEARCH =====
function searchTranscript(q){
  if(!q||q.length<2){document.getElementById('transres').innerHTML='';return;}
  const s=DB.students.find(x=>(x.first+' '+x.last+' '+x.reg).toLowerCase().includes(q.toLowerCase()));
  if(!s){document.getElementById('transres').innerHTML='<div class="alert ad">No student found matching that query.</div>';return;}
  const mks=DB.marks.filter(m=>m.stu_id==s.id);
  const gpa=mks.length?((mks.reduce((a,m)=>a+m.gpa,0)/mks.length).toFixed(2)):'-';
